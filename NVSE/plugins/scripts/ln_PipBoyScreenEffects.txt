; Pip-Boy Screen Effects - Stentorious

; On game load
Goo1.SetOnTimerUpdateHandler 1 (CompileScript "PipBoyScreenEffects\EffectEasing.gek") "*PipScreen:Blood"
Goo1.SetOnTimerUpdateHandler 1 (CompileScript "PipBoyScreenEffects\EffectEasing.gek") "*PipScreen:Crack"
Goo1.SetOnTimerUpdateHandler 1 (CompileScript "PipBoyScreenEffects\EffectEasing.gek") "*PipScreen:Water"
Goo1.SetOnTimerUpdateHandler 1 (CompileScript "PipBoyScreenEffects\EffectEasing.gek") "*PipScreen:Dust"
Goo1.SetOnTimerStopHandler 1 (CompileScript "PipBoyScreenEffects\EffectTimerEnd.gek") "*PipScreen:Blood"
Goo1.SetOnTimerStopHandler 1 (CompileScript "PipBoyScreenEffects\EffectTimerEnd.gek") "*PipScreen:Crack"
Goo1.SetOnTimerStopHandler 1 (CompileScript "PipBoyScreenEffects\EffectTimerEnd.gek") "*PipScreen:Water"
Goo1.SetOnTimerStopHandler 1 (CompileScript "PipBoyScreenEffects\EffectTimerEnd.gek") "*PipScreen:Dust"
Goo1.SetOnTimerStopHandler 1 (CompileScript "PipBoyScreenEffects\HitTimerEnd.gek") "*PipScreen:Hit"
CallAfterFrames 2 (CompileScript "PipBoyScreenEffects\UpdateTextureLoad.gek")

; On game restart
if Goo1.AuxVarGetFlt "*PipScreen_Init" == 1
	return
endif
Goo1.AuxVarSetFlt "*PipScreen_Init" 1

; Check for requirements
if GetNVSEVersionFull < 6.40
	MessageBoxEx "Pip-Boy Screen Effects missing requirement!%rInstall xNVSE 6.4.0+."
	return
endif
if GetPluginVersion "JohnnyGuitarNVSE" < 390
	MessageBoxEx "Pip-Boy Screen Effects missing requirement!%rInstall latest JohnnyGuitar NVSE plugin."
	return
endif
if GetPluginVersion "ShowOffNVSE Plugin" < 180
	MessageBoxEx "Pip-Boy Screen Effects missing requirement!%rInstall latest ShowOff NVSE plugin."
	return
endif

; Load effects settings
; "*PipScreen_Effect"
; 0 : Enable
; 1 : Brightness
; 2 : Fade time
; 3 : Extra config
; 4 : Effect state
; 5 : Applied fade time

Goo1.AuxVarSetFlt "*PipScreen:Blood" (GetINIFloat_Cached "Blood:bEnable" "Stentorious\PipBoyScreenEffects.ini" 1) 0
Goo1.AuxVarSetFlt "*PipScreen:Blood" (GetINIFloat_Cached "Blood:fBrightness" "Stentorious\PipBoyScreenEffects.ini" 1) 1
Goo1.AuxVarSetFlt "*PipScreen:Blood" (GetINIFloat_Cached "Blood:fFadeTime" "Stentorious\PipBoyScreenEffects.ini" 30) 2
Goo1.AuxVarSetFlt "*PipScreen:Blood" (GetINIFloat_Cached "Blood:fThreshold" "Stentorious\PipBoyScreenEffects.ini" 80) 3
Goo1.AuxVarSetFlt "*PipScreen:Blood" 0 4
Goo1.AuxVarSetFlt "*PipScreen:Blood" (Goo1.AuxVarGetFlt "*PipScreen:Blood" 2) 5
Goo1.AuxVarSetFlt "*PipScreen:Water" (GetINIFloat_Cached "Water:bEnable" "Stentorious\PipBoyScreenEffects.ini" 1) 0
Goo1.AuxVarSetFlt "*PipScreen:Water" (GetINIFloat_Cached "Water:fBrightness" "Stentorious\PipBoyScreenEffects.ini" 1) 1
Goo1.AuxVarSetFlt "*PipScreen:Water" (GetINIFloat_Cached "Water:fFadeTime" "Stentorious\PipBoyScreenEffects.ini" 10) 2
Goo1.AuxVarSetFlt "*PipScreen:Water" (GetINIFloat_Cached "Water:fWashTime" "Stentorious\PipBoyScreenEffects.ini" 5) 3
Goo1.AuxVarSetFlt "*PipScreen:Water" 0 4
Goo1.AuxVarSetFlt "*PipScreen:Water" (Goo1.AuxVarGetFlt "*PipScreen:Water" 2) 5
Goo1.AuxVarSetFlt "*PipScreen:Crack" (GetINIFloat_Cached "Crack:bEnable" "Stentorious\PipBoyScreenEffects.ini" 1) 0
Goo1.AuxVarSetFlt "*PipScreen:Crack" (GetINIFloat_Cached "Crack:fBrightness" "Stentorious\PipBoyScreenEffects.ini" 1) 1
Goo1.AuxVarSetFlt "*PipScreen:Crack" 0 2
Goo1.AuxVarSetFlt "*PipScreen:Crack" (GetINIFloat_Cached "Crack:fThreshold" "Stentorious\PipBoyScreenEffects.ini" 30) 3
Goo1.AuxVarSetFlt "*PipScreen:Crack" 0 4
Goo1.AuxVarSetFlt "*PipScreen:Crack" (Goo1.AuxVarGetFlt "*PipScreen:Crack" 2) 5
Goo1.AuxVarSetFlt "*PipScreen:Dust" (GetINIFloat_Cached "Dust:bEnable" "Stentorious\PipBoyScreenEffects.ini" 1) 0
Goo1.AuxVarSetFlt "*PipScreen:Dust" (GetINIFloat_Cached "Dust:fBrightness" "Stentorious\PipBoyScreenEffects.ini" 1) 1
Goo1.AuxVarSetFlt "*PipScreen:Dust" (GetINIFloat_Cached "Dust:fFadeTime" "Stentorious\PipBoyScreenEffects.ini" 30) 2
Goo1.AuxVarSetFlt "*PipScreen:Dust" 0 3
Goo1.AuxVarSetFlt "*PipScreen:Dust" 0 4
Goo1.AuxVarSetFlt "*PipScreen:Dust" (Goo1.AuxVarGetFlt "*PipScreen:Dust" 2) 5
CloseFileSO "Stentorious\PipBoyScreenEffects.ini" 0

; Sandstorms support
AssignKeyword HVDustStorm "WeatherSandstorm"
if eval IsModLoaded "Desert Natural Sandstorms.esp" || IsModLoaded "Desert Natural Sandstorms TTW.esp"
	AssignKeyword (EditorIDToFormID "SandstormWeather") "WeatherSandstorm"
	AssignKeyword (EditorIDToFormID "SandstormWeather2") "WeatherSandstorm"
endif
if eval IsModLoaded "SSTAtmospheresWeather.esp"
	AssignKeyword (EditorIDToFormID "SSTNVWastelandSandstorm") "WeatherSandstorm"
endif

; Event handlers
SGMLC (CompileScript "PipBoyScreenEffects\UpdateTexture.gek") 1 60 3
SetEventHandler "OnEquip" ({ref rEquipper, ref rEquipped} => CallAfterFrames 0 (CompileScript "PipBoyScreenEffects\UpdateEquip.gek") 1) "first"::Player
SetEventHandler "OnUnequip" ({ref rEquipper, ref rEquipped} => CallAfterFrames 0 (CompileScript "PipBoyScreenEffects\UpdateEquip.gek") 1) "first"::Player
SetOnHitEventHandler (CompileScript "PipBoyScreenEffects\OnHit.gek") 1 Player
SetOnFastTravelEventHandler ({ref rMapMarker} => Call (CompileScript "PipBoyScreenEffects\UpdateTextureLoad.gek")) 1

; Pip-Boy Screen Always On support
if eval IsModLoaded "Pip-Boy Screen Always On.esm"
	Goo1.AuxVarSetFlt "*PipScreen:AlwaysOn" 1
elseif eval FileExists "nvse\plugins\scripts\gr_pipboy_screen_always_on.txt"
	Goo1.AuxVarSetFlt "*PipScreen:AlwaysOn" 2
endif

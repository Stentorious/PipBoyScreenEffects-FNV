; Pip-Boy Screen Effects - Stentorious

; On game load
SetOnTimerUpdateHandler 1 (CompileScript "PipBoyScreenEffects\EffectEasing.gek") "*PipScreen:Blood"
SetOnTimerUpdateHandler 1 (CompileScript "PipBoyScreenEffects\EffectEasing.gek") "*PipScreen:Crack"
SetOnTimerUpdateHandler 1 (CompileScript "PipBoyScreenEffects\EffectEasing.gek") "*PipScreen:Water"
SetOnTimerUpdateHandler 1 (CompileScript "PipBoyScreenEffects\EffectEasing.gek") "*PipScreen:Dust"
SetOnTimerStopHandler 1 (CompileScript "PipBoyScreenEffects\EffectTimerEnd.gek") "*PipScreen:Blood"
SetOnTimerStopHandler 1 (CompileScript "PipBoyScreenEffects\EffectTimerEnd.gek") "*PipScreen:Crack"
SetOnTimerStopHandler 1 (CompileScript "PipBoyScreenEffects\EffectTimerEnd.gek") "*PipScreen:Water"
SetOnTimerStopHandler 1 (CompileScript "PipBoyScreenEffects\EffectTimerEnd.gek") "*PipScreen:Dust"
CallAfterFrames 2 (CompileScript "PipBoyScreenEffects\UpdateTextureLoad.gek")

; On game restart
if Goo1.AuxVarGetFlt "*PipScreen_Init" == 1
	return
endif
Goo1.AuxVarSetFlt "*PipScreen_Init" 1

; Check for requirements
if GetNVSEVersionFull < 6.40
	MessageBoxEx "Pip-Boy Screen Effects missing requirement!%rInstall xNVSE 6.4.0+."
	return
endif
if GetPluginVersion "JohnnyGuitarNVSE" < 390
	MessageBoxEx "Pip-Boy Screen Effects missing requirement!%rInstall latest JohnnyGuitar NVSE plugin."
	return
endif
if GetPluginVersion "ShowOffNVSE Plugin" < 145
	MessageBoxEx "Pip-Boy Screen Effects missing requirement!%rInstall latest ShowOff NVSE plugin."
	return
endif

; Load INI settings
Goo1.AuxVarSetFlt "*PipScreen_INI" (GetINIFloat_Cached "Conditions:fHealthThreshold" "Stentorious\PipBoyScreenEffects.ini" 80) 0
Goo1.AuxVarSetFlt "*PipScreen_INI" (GetINIFloat_Cached "Conditions:fArmorConditionThreshold" "Stentorious\PipBoyScreenEffects.ini" 30) 1

; Load effects settings
; "*PipScreen_Effect"
; 0 : Toggle
; 1 : Brightness
; 2 : Fade time
; 3 : General state (0 = Hidden, 1 = Visible, 2 = Fade in, 3 = Fade out)
; 4 : Fade state (0 = Hidden, 1 = Visible, 2 = Fade in, 3 = Fade out)

Goo1.AuxVarSetFlt "*PipScreen:Blood" (GetINIFloat_Cached "Effects:bBlood" "Stentorious\PipBoyScreenEffects.ini" 1) 0
Goo1.AuxVarSetFlt "*PipScreen:Blood" (GetINIFloat_Cached "Effects:fBloodBrightness" "Stentorious\PipBoyScreenEffects.ini" 1) 1
Goo1.AuxVarSetFlt "*PipScreen:Crack" (GetINIFloat_Cached "Effects:bCrack" "Stentorious\PipBoyScreenEffects.ini" 1) 0
Goo1.AuxVarSetFlt "*PipScreen:Crack" (GetINIFloat_Cached "Effects:fCrackBrightness" "Stentorious\PipBoyScreenEffects.ini" 1) 1
Goo1.AuxVarSetFlt "*PipScreen:Water" (GetINIFloat_Cached "Effects:bWater" "Stentorious\PipBoyScreenEffects.ini" 1) 0
Goo1.AuxVarSetFlt "*PipScreen:Water" (GetINIFloat_Cached "Effects:fWaterBrightness" "Stentorious\PipBoyScreenEffects.ini" 1) 1

Goo1.AuxVarSetFlt "*PipScreen:Dust" (GetINIFloat_Cached "Effects:bDust" "Stentorious\PipBoyScreenEffects.ini" 1) 0
Goo1.AuxVarSetFlt "*PipScreen:Dust" (GetINIFloat_Cached "Effects:fDustBrightness" "Stentorious\PipBoyScreenEffects.ini" 1) 1
CloseFileSO "Stentorious\PipBoyScreenEffects.ini" 0

; Load default time settings
Goo1.AuxVarSetFlt "*PipScreen:Blood" 30 2
Goo1.AuxVarSetFlt "*PipScreen:Crack" 1 2
Goo1.AuxVarSetFlt "*PipScreen:Water" 10 2
Goo1.AuxVarSetFlt "*PipScreen:Dust" 30 2

; Sandstorms support
AssignKeyword HVDustStorm "WeatherSandstorm"
if eval IsModLoaded "Desert Natural Sandstorms.esp" || IsModLoaded "Desert Natural Sandstorms TTW.esp"
	AssignKeyword (EditorIDToFormID "SandstormWeather") "WeatherSandstorm"
	AssignKeyword (EditorIDToFormID "SandstormWeather2") "WeatherSandstorm"
endif
if eval IsModLoaded "SSTAtmospheresWeather.esp"
	AssignKeyword (EditorIDToFormID "SSTNVWastelandSandstorm") "WeatherSandstorm"
endif

; Pip-Boy support
Goo1.AuxVarSetFlt "*PipScreen_INI" (eval IsModLoaded "pipboy2500_edisleado.esp") 2
if eval Goo1.AuxVarGetFlt "*PipScreen_INI" 2
	AttachModel (EditorIDToFormID "BIDEpipReadius") 3 "glare|PipBoyScreenEffects\2500.nif"
endif

; Update events
SGMLC (CompileScript "PipBoyScreenEffects\UpdateTexture.gek") 1 60 3
SetEventHandler "OnEquip" ({ref rEquipper, ref rEquipped} => CallAfterFrames 0 (CompileScript "PipBoyScreenEffects\UpdateEquip.gek") 1) "first"::Player
SetEventHandler "OnUnequip" ({ref rEquipper, ref rEquipped} => CallAfterFrames 0 (CompileScript "PipBoyScreenEffects\UpdateEquip.gek") 1) "first"::Player
SetOnHitEventHandler (CompileScript "PipBoyScreenEffects\OnHit.gek") 1 Player
SetOnFastTravelEventHandler ({ref rMapMarker} => Call (CompileScript "PipBoyScreenEffects\UpdateTextureLoad.gek")) 1
